\section{Recubrimientos normales}
Sea $f \colon Y \to X$ un recubrimiento.
Recuerde que una \strong{transformación de cubrierta} (eng.\ \emph{deck-transformation}) es un homeomorfismo $g \colon Y \to Y$ tal que el
diagrama conmuta:
% https://q.uiver.app/#q=WzAsMyxbMCwwLCJZIl0sWzIsMCwiWSJdLFsxLDEsIlgiXSxbMCwxLCJcXHNpbSIsMl0sWzAsMiwiZiIsMl0sWzEsMiwiZiJdXQ==
\[\begin{tikzcd}
	Y && Y \\
	& X
	\arrow["g", "\sim"', from=1-1, to=1-3]
	\arrow["f"', from=1-1, to=2-2]
	\arrow["f", from=1-3, to=2-2]
\end{tikzcd}\]
Es inmediato que éstas conforman un grupo, denotado $\Aut(Y/X)$ ó $\Aut_X(Y)$.
\begin{mydef}
	Diremos que un recubrimiento $f\colon Y \to X$ es \strong{normal} (o \emph{regular}) si para cada par de puntos $y, y' \in Y$ con
	$f(y) = f(y')$, existe una transformación de cubierta $g \in \Aut(Y/X)$ tal que $g(y) = y'$.
\end{mydef}
\begin{enumerate}
	\item \textbf{Prueba de lucidez:}
		Sea $G \acts X$ la acción de un grupo sobre un espacio que es libre y propiamente discontinua.
		Pruebe que el recubrimiento $X \epicto G\coquot X$ es normal.

	\item\label{ex:twofold_cov}
		Pruebe que todo recubrimiento de dos hojas es normal.

	\item\lookright Sea $f \colon Y \to X$ un recubrimiento normal de un espacio arcoconexo.
		Muestre que, para todo $y\in Y$ con $x := f(y)$, el homomorfismo $\pi_1(f, y) \colon \pi_1(Y, y) \to \pi_1(X, x)$ es
		inyectivo e identifica a $\pi_1(Y, y)$ con un subgrupo normal.
		Más aún,
		\[
			\pi_1(X, x) / f_*\pi_1(Y, y) \cong \Aut(Y/X).
		\]

\section{Recubrimientos universales}
\begin{enumerate}[resume]
	\item \textbf{Existencia de recubrimiento universal:}
		Sea $X$ un espacio arcoconexo.
		Se dice que $f\colon Y \to X$ es un \emph{recubrimiento universal} si $f$ es recubrimiento e $Y$ es simplemente conexo.
		\begin{enumerate}
			\item Pruebe que si $X$ posee un recubrimiento universal, entonces es único salvo homeomorfismo (relativo a $X$).
			\item\lookst
				Pruebe que $X$ posee algún recubrimiento universal cuando todo punto $x$ posee un entorno arcoconexo $U$ tal
				que todo lazo basado en $x$ contenido en $U$ es contráctil \underline{en} $X$ (en cuyo caso, decimos que $X$
				es \emph{semilocalmente simplemente conexo}).
			\item\lookup
				Dé un ejemplo de un espacio que no sea semilocalmente simplemente conexo.
		\end{enumerate}

	\item Sea $f \colon \tilde X \to X$ el recubrimiento universal de un espacio arcoconexo $X$ (si existe).
		Muestre que $f$ es un recubrimiento normal y que $\Aut(\tilde X/X) \cong \pi_1(X, x)$ para cualquier punto $x \in X$.
\end{enumerate}
